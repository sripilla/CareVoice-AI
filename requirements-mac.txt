# CareVoice AI on Apple Silicon (M1/M2/M3, including M2 Ultra).
#
#   python3.11 -m venv .venv
#   source .venv/bin/activate
#   pip install -r requirements-mac.txt
#
# Then always export this before running anything -- MPS does not
# implement every PyTorch op and needs a CPU fallback path:
#
#   export PYTORCH_ENABLE_MPS_FALLBACK=1
#
# Differences from requirements.txt:
#
#   * Versions are floors, not exact pins. The main requirements file
#     is a frozen x86/Windows environment; pinning those exact builds
#     on arm64 macOS resolves badly.
#   * torch comes from PyPI, which ships arm64 wheels with MPS built
#     in. Do not install a CUDA build.
#   * ctranslate2/faster-whisper are still here for the `cascade`
#     pipeline, but note they are CPU-only on macOS -- CTranslate2
#     has no Metal backend. Use CAREVOICE_PIPELINE=seamless for
#     GPU-accelerated speech.
